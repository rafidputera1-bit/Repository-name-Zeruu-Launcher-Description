package com.zeruu.launcher.data

import androidx.compose.runtime.Immutable
import java.util.UUID

@Immutable
sealed interface HomeItem {
    val id: String
    val page: Int
    val col: Int
    val row: Int
    val spanX: Int
    val spanY: Int

    fun moveTo(page: Int, col: Int, row: Int): HomeItem
    fun resizeTo(spanX: Int, spanY: Int): HomeItem

    @Immutable
    data class AppShortcut(
        override val id: String = UUID.randomUUID().toString(),
        val componentKey: String,
        override val page: Int,
        override val col: Int,
        override val row: Int,
        override val spanX: Int = 1,
        override val spanY: Int = 1
    ) : HomeItem {
        override fun moveTo(page: Int, col: Int, row: Int) = copy(page = page, col = col, row = row)
        override fun resizeTo(spanX: Int, spanY: Int) = copy(spanX = spanX, spanY = spanY)
    }

    @Immutable
    data class Folder(
        override val id: String = UUID.randomUUID().toString(),
        val title: String = "Folder",
        val appKeys: List<String>,
        override val page: Int,
        override val col: Int,
        override val row: Int,
        override val spanX: Int = 1,
        override val spanY: Int = 1
    ) : HomeItem {
        override fun moveTo(page: Int, col: Int, row: Int) = copy(page = page, col = col, row = row)
        override fun resizeTo(spanX: Int, spanY: Int) = copy(spanX = spanX, spanY = spanY)
    }

    @Immutable
    data class Widget(
        override val id: String = UUID.randomUUID().toString(),
        val appWidgetId: Int,
        val providerPackage: String,
        val providerClass: String,
        override val page: Int,
        override val col: Int,
        override val row: Int,
        override val spanX: Int = 2,
        override val spanY: Int = 2
    ) : HomeItem {
        override fun moveTo(page: Int, col: Int, row: Int) = copy(page = page, col = col, row = row)
        override fun resizeTo(spanX: Int, spanY: Int) = copy(spanX = spanX, spanY = spanY)
    }
}

package com.zeruu.launcher.data

import android.content.Context
import org.json.JSONArray
import org.json.JSONObject

class HomeLayoutRepository(context: Context) {

    private val prefs = context.applicationContext.getSharedPreferences("zeruu_layout_v5", Context.MODE_PRIVATE)
    private val legacyPrefs = context.applicationContext.getSharedPreferences("zeruu_layout", Context.MODE_PRIVATE)

    fun hasLayout(): Boolean = prefs.contains("items") || legacyPrefs.contains("home")

    fun hasV5Layout(): Boolean = prefs.contains("items")

    fun loadLegacyHome(): List<String> {
        val s = legacyPrefs.getString("home", "") ?: ""
        return if (s.isEmpty()) emptyList() else s.split("\n")
    }

    fun loadLegacyDock(): List<String> {
        val s = legacyPrefs.getString("dock", "") ?: ""
        return if (s.isEmpty()) emptyList() else s.split("\n")
    }

    fun loadDock(): List<String> {
        val s = prefs.getString("dock", null) ?: return loadLegacyDock()
        return if (s.isEmpty()) emptyList() else s.split("\n")
    }

    fun loadSeen(): Set<String> = prefs.getStringSet("seen", null)?.toSet()
        ?: legacyPrefs.getStringSet("seen", emptySet())?.toSet()
        ?: emptySet()

    fun loadItems(): List<HomeItem> {
        val raw = prefs.getString("items", null) ?: return emptyList()
        return runCatching {
            val arr = JSONArray(raw)
            buildList {
                for (i in 0 until arr.length()) {
                    val o = arr.getJSONObject(i)
                    val id = o.getString("id")
                    val page = o.getInt("page")
                    val col = o.getInt("col")
                    val row = o.getInt("row")
                    val spanX = o.optInt("spanX", 1).coerceAtLeast(1)
                    val spanY = o.optInt("spanY", 1).coerceAtLeast(1)
                    when (o.getString("type")) {
                        "app" -> add(HomeItem.AppShortcut(id, o.getString("key"), page, col, row, spanX, spanY))
                        "folder" -> {
                            val keysArr = o.getJSONArray("keys")
                            val keys = List(keysArr.length()) { k -> keysArr.getString(k) }
                            add(HomeItem.Folder(id, o.optString("title", "Folder"), keys, page, col, row, spanX, spanY))
                        }
                        "widget" -> add(
                            HomeItem.Widget(
                                id, o.getInt("widgetId"), o.getString("pkg"), o.getString("cls"),
                                page, col, row, spanX, spanY
                            )
                        )
                    }
                }
            }
        }.getOrElse { emptyList() }
    }

    fun save(items: List<HomeItem>, dock: List<String>, seen: Set<String>) {
        val arr = JSONArray()
        items.forEach { item ->
            val o = JSONObject()
                .put("id", item.id)
                .put("page", item.page)
                .put("col", item.col)
                .put("row", item.row)
                .put("spanX", item.spanX)
                .put("spanY", item.spanY)
            when (item) {
                is HomeItem.AppShortcut -> o.put("type", "app").put("key", item.componentKey)
                is HomeItem.Folder -> o.put("type", "folder").put("title", item.title).put("keys", JSONArray(item.appKeys))
                is HomeItem.Widget -> o.put("type", "widget")
                    .put("widgetId", item.appWidgetId)
                    .put("pkg", item.providerPackage)
                    .put("cls", item.providerClass)
            }
            arr.put(o)
        }
        prefs.edit()
            .putString("items", arr.toString())
            .putString("dock", dock.joinToString("\n"))
            .putStringSet("seen", seen)
            .apply()
    }
}
